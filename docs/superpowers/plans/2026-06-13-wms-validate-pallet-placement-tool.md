```bsl
//+Лико m.shenderov 13.06.2026 — Валидация размещения паллета в ячейку
// Возвращает Структуру:
//   Ячейка  — ссылка на СкладскуюЯчейку (или Неопределено если не найдена)
//   Паллет  — ссылка на Лико_Паллеты2_0 (или Неопределено если не найден)
//   Ошибки  — Массив строк (пустой если размещение валидно)
//   Валидно — Булево (Истина = ошибок нет)
&НаСервере
Функция ОшибкиРазмещенияПаллета(КодЯчейки, КодПаллета) Экспорт
	
	Результат = Новый Структура;
	Результат.Вставить("Ячейка",  Неопределено);
	Результат.Вставить("Паллет",  Неопределено);
	Результат.Вставить("Ошибки",  Новый Массив);
	Результат.Вставить("Валидно", Ложь);
	
	// 1. Поиск ячейки: код → наименование
	Ячейка = Справочники.СкладскиеЯчейки.НайтиПоКоду(КодЯчейки);
	Если Не ЗначениеЗаполнено(Ячейка) Тогда
		Ячейка = Справочники.СкладскиеЯчейки.НайтиПоНаименованию(КодЯчейки);
	КонецЕсли;
	
	Если Не ЗначениеЗаполнено(Ячейка) Тогда
		Результат.Ошибки.Добавить("Ячейка не найдена по коду/наименованию: """ + КодЯчейки + """");
		Возврат Результат;
	КонецЕсли;
	Результат.Ячейка = Ячейка;
	
	// 2. Поиск паллета: код → наименование
	Паллет = Справочники.Лико_Паллеты2_0.НайтиПоКоду(КодПаллета);
	Если Не ЗначениеЗаполнено(Паллет) Тогда
		Паллет = Справочники.Лико_Паллеты2_0.НайтиПоНаименованию(КодПаллета);
	КонецЕсли;
	
	Если Не ЗначениеЗаполнено(Паллет) Тогда
		Результат.Ошибки.Добавить("Паллет не найден по коду/наименованию: """ + КодПаллета + """");
		Возврат Результат;
	КонецЕсли;
	Результат.Паллет = Паллет;
	
	// 3. Вызов существующей валидации
	Результат.Ошибки = Справочники.Лико_СкладскиеСекции.ОшибкиРазмещенияПаллетаВАдрес(Ячейка, Паллет);
	Результат.Валидно = (Результат.Ошибки.Количество() = 0);
	
	Возврат Результат;
	
КонецФункции //-<---Лико m.shenderov 13.06.2026
```

### Task 3: Адаптер `Адаптер_ОшибкиРазмещенияПаллета`

**Файл:** `1s/ERP/obrab/ZagSklad/ЗагрузкаСклада/Forms/Форма/Ext/Form/Module.bsl` — добавить в диспетчер-адаптер, после `Адаптер_ЧемпионПоФутболу` (строка ~2895)

```bsl
//+Лико m.shenderov 13.06.2026 — Адаптер: аргументы LLM → ОшибкиРазмещенияПаллета → JSON
&НаСервере
Функция Адаптер_ОшибкиРазмещенияПаллета(Аргументы)
	КодЯчейки  = "";
	КодПаллета = "";
	Если ТипЗнч(Аргументы) = Тип("Структура") Тогда
		Если Аргументы.Свойство("cell")   Тогда КодЯчейки  = Аргументы.cell;   КонецЕсли;
		Если Аргументы.Свойство("pallet") Тогда КодПаллета = Аргументы.pallet; КонецЕсли;
	КонецЕсли;
	
	Если Не ЗначениеЗаполнено(КодЯчейки) Или Не ЗначениеЗаполнено(КодПаллета) Тогда
		Возврат СериализоватьJSON(Новый Структура("error",
			"Не указаны обязательные параметры: cell и pallet"));
	КонецЕсли;
	
	Данные = ОшибкиРазмещенияПаллета(КодЯчейки, КодПаллета);
	
	// Формируем JSON-ответ
	Ответ = Новый Структура;
	Ответ.Вставить("valid",     Данные.Валидно);
	Ответ.Вставить("cell",      Строка(Данные.Ячейка));
	Ответ.Вставить("pallet",    Строка(Данные.Паллет));
	Ответ.Вставить("errors",    Данные.Ошибки);
	Ответ.Вставить("cellFound", ЗначениеЗаполнено(Данные.Ячейка));
	Ответ.Вставить("palletFound", ЗначениеЗаполнено(Данные.Паллет));
	
	Если Данные.Валидно Тогда
		Ответ.Вставить("message", "Размещение валидно. Паллет " + КодПаллета
			+ " можно разместить в ячейку " + КодЯчейки + ".");
	ИначеЕсли Данные.Ошибки.Количество() > 0 Тогда
		Ответ.Вставить("message", СтрСоединить(Данные.Ошибки, "; "));
	КонецЕсли;
	
	Возврат СериализоватьJSON(Ответ);
КонецФункции //-<---Лико m.shenderov 13.06.2026
```
